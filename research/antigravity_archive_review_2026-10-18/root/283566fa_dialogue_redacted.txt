_swap_cache = {"value": "0M", "counter": 0}

def get_sys_mem_str() -> str:
    """Returns Apple Silicon Metal unified memory and swap usage string."""
    global _swap_cache
    try:
        active_gb = mx.get_active_memory() / 1e9
        peak_gb = mx.get_peak_memory() / 1e9
        _swap_cache["counter"] += 1
        # Refresh sysctl swap usage only every 5th log call to avoid subprocess overhead
        if _swap_cache["counter"] % 5 == 1:
            swap_res = subprocess.run(["sysctl", "vm.swapusage"], capture_output=True, text=True)
            swap_parts = swap_res.stdout.strip().split()
            _swap_cache["value"] = swap_parts[6] if len(swap_parts) >= 7 else "0M"
        return f"Metal Unified GPU: {active_gb:.2f}GB (Peak: {peak_gb:.2f}GB) | Swap: {_swap_cache['value']}"
    except Exception:
        return ""

def clip_grad_norm_mlx(grads, max_norm: float = 1.0, scale: float = 1.0):
    """Clips global gradient L2 norm to max_norm and optionally divides by scale.
    Fuses accumulation scaling + clipping into a single tree traversal."""
    total_norm_sq = mx.array(0.0, dtype=mx.float32)
    for _, g in tree_flatten(grads):
        total_norm_sq = total_norm_sq + mx.sum(g.astype(mx.float32) ** 2)
    total_norm = mx.sqrt(total_norm_sq)
    # Compute effective norm after scaling: ||g / scale||₂ = ||g||₂ / scale
    effective_norm = total_norm / scale if scale != 1.0 else total_norm
    clip_coef = max_norm / (effective_norm + 1e-6)
    # Fused coefficient: (1 / scale) * min(1, clip_coef)
    combined = mx.minimum(mx.array(1.0, dtype=mx.float32), clip_coef) / scale
    clipped = tree_map(lambda g: g * combined.astype(g.dtype), grads)
    return clipped, effective_norm
```

class MLXBlock(nn.Module):
    def __init__(self, d_model: int, n_heads: int, n_kv_heads: int):
        super().__init__()
        self.norm1 = MLXRMSNorm(d_model)
        self.norm2 = MLXRMSNorm(d_model)
        self.head_dim = d_model // n_heads
        self.n_heads = n_heads
        self.n_kv_heads = n_kv_heads
        self.kv_groups = n_heads // n_kv_heads

        self.qkv_proj = nn.Linear(d_model, (n_heads + 2 * n_kv_heads) * self.head_dim, bias=False)
        self.out = nn.Linear(d_model, d_model, bias=False)
        self.mlp = MLXSwiGLU(d_model)
        
        # Precompute QKV split boundaries (static across all forward passes)
        self._q_end = n_heads * self.head_dim
        self._k_end = self._q_end + n_kv_heads * self.head_dim

    def __call__(self, x):
        B, T, D = x.shape
        h = self.norm1(x)

        qkv = self.qkv_proj(h)
        # Split using precomputed static index boundaries
        q, k, v = mx.split(qkv, [self._q_end, self._k_end], axis=-1)
        q = q.reshape(B, T, self.n_heads, self.head_dim).transpose(0, 2, 1, 3)
        k = k.reshape(B, T, self.n_kv_heads, self.head_dim).transpose(0, 2, 1, 3)
        v = v.reshape(B, T, self.n_kv_heads, self.head_dim).transpose(0, 2, 1, 3)

        # Apply hardware-accelerated RoPE Metal kernel to Q and K
        q = mx.fast.rope(q, self.head_dim, traditional=False, base=10000.0, scale=1.0, offset=0)
        k = mx.fast.rope(k, self.head_dim, traditional=False, base=10000.0, scale=1.0, offset=0)

        scale = 1.0 / (self.head_dim ** 0.5)
        out = mx.fast.scaled_dot_product_attention(q, k, v, scale=scale)
        out = out.transpose(0, 2, 1, 3).reshape(B, T, D)

        x = x + self.out(out)
        x = x + self.mlp(self.norm2(x))
        return x
```

class COROSredBlock(nn.Module):
    """Transformer block with dynamic Causal (Draft) vs. Bidirectional (Refine) attention."""

    def __init__(self, d_model: int, n_heads: int, n_kv_heads: int):
        super().__init__()
        self.norm1 = MLXRMSNorm(d_model)
        self.norm2 = MLXRMSNorm(d_model)
        self.head_dim = d_model // n_heads
        self.n_heads = n_heads
        self.n_kv_heads = n_kv_heads
        self.kv_groups = n_heads // n_kv_heads

        # Fused QKV projection for single-dispatch Metal execution
        self.qkv_proj = nn.Linear(d_model, (n_heads + 2 * n_kv_heads) * self.head_dim, bias=False)
        self.out = nn.Linear(d_model, d_model, bias=False)
        self.mlp = MLXSwiGLU(d_model)
        self._mask_cache = {}
        
        self._q_end = n_heads * self.head_dim
        self._k_end = self._q_end + n_kv_heads * self.head_dim

    def __call__(self, x, is_causal: bool = True):
        B, T, D = x.shape
        h = self.norm1(x)

        qkv = self.qkv_proj(h)
        q, k, v = mx.split(qkv, [self._q_end, self._k_end], axis=-1)
        q = q.reshape(B, T, self.n_heads, self.head_dim).transpose(0, 2, 1, 3)
        k = k.reshape(B, T, self.n_kv_heads, self.head_dim).transpose(0, 2, 1, 3)
        v = v.reshape(B, T, self.n_kv_heads, self.head_dim).transpose(0, 2, 1, 3)

        # Apply rotary position embeddings using Apple Metal accelerated kernel
        q = mx.fast.rope(q, self.head_dim, traditional=False, base=10000.0, scale=1.0, offset=0)
        k = mx.fast.rope(k, self.head_dim, traditional=False, base=10000.0, scale=1.0, offset=0)

        # Build or retrieve lower-triangular causal mask only when in draft mode
        mask = None
        if is_causal:
            cache_key = (T, str(q.dtype))
            if cache_key not in self._mask_cache:
                self._mask_cache[cache_key] = nn.MultiHeadAttention.create_additive_causal_mask(T).astype(q.dtype)
            mask = self._mask_cache[cache_key]

        scale = 1.0 / (self.head_dim ** 0.5)
        out = mx.fast.scaled_dot_product_attention(q, k, v, scale=scale, mask=mask)
        out = out.transpose(0, 2, 1, 3).reshape(B, T, D)

        x = x + self.out(out)
        x = x + self.mlp(self.norm2(x))
        return x
```

    def refine(self, sequence: list[int], flagged_indices: list[int]) -> list[int]:
        """
        Phase 3: Selectively masks flagged tokens and performs bidirectional MDLM refinement passes.
        """
        if not flagged_indices:
            return sequence

        refined = list(sequence)
        for idx in flagged_indices:
            refined[idx] = self.mask_token_id

        flagged_mx = mx.array(flagged_indices)

        for _ in range(self.refine_steps):
            inp = mx.array([refined], dtype=mx.int32)
            logits = self.model(inp, is_causal=False, return_reliability=False)

            # Vectorized multi-position argmax in one single Metal call
            flagged_logits = logits[0, flagged_mx]
            best_tokens = mx.argmax(flagged_logits, axis=-1)
            mx.eval(best_tokens)
            best_np = np.array(best_tokens)
            for i, idx in enumerate(flagged_indices):
                refined[idx] = int(best_np[i])

        return refined
```

The PyTorch-XLA training pipeline in [Unified_Training_Suite.ipynb](file:///Users/ivansamuel/Documents/Projects/telos/notebooks/shared/Unified_Training_Suite.ipynb) and [Unified_Training_Suite.py](file:///Users/ivansamuel/Documents/Projects/telos/Unified_Training_Suite.py) is tailored to the TPU v5e-8 architecture across five key dimensions:

---

### 1. Hardware Matrix Multiplier (MXU) Alignment
TPU v5e compute cores utilize **$128 \times 128$ systolic arrays** (Matrix Multiply Units). Any tensor dimension not divisible by 128 results in zero-padding overhead in hardware registers:
- **Vocabulary Size**: `8192` ($64 \times 128$) — exact MXU alignment with zero padding penalty.
- **Sequence Length**: `512` ($4 \times 128$) — exact MXU alignment.
- **Hidden Dimensions**:
  - **12.5M**: $d=256$ ($2 \times 128$)
  - **25M**: $d=384$ ($3 \times 128$)
  - **50M**: $d=512$ ($4 \times 128$)
- **SwiGLU Expansion**: Explicitly rounded to multiples of 64/128 (`64 * ((hidden_dim + 63) // 64)`).

---

### 2. PyTorch-XLA SPMD Mesh (Single Program Multiple Data)
Rather than using legacy multi-process `xmp.spawn` (which incurs IPC latency and Python process overhead), the suite uses native **XLA GSPMD**:
```python
xr.use_spmd()
_num_tpu_devices = xr.global_runtime_device_count()  # 8 chips on v5e-8
_tpu_mesh = Mesh(np.arange(_num_tpu_devices), (_num_tpu_devices,), ('data',))

# In training loop:
xs.mark_sharding(x, mesh, ('data', None))  # Zero-copy batch sharding
```
The XLA compiler automatically handles data-parallel sharding across all 8 TPU v5e chips via Google’s fast Inter-Core Interconnect (ICI).

---

### 3. Execution Profile on v5e-8
Configured in `configs/unified/25m/telos_25m_r25.yaml`:
- **Per-Chip Micro-Batch**: `batch_size: 48`
- **Number of Devices**: `num_devices: 8`
- **Global Effective Batch**: $48 \times 8 = 384$ sequences ($196,608$ tokens/step)
- **Gradient Accumulation**: `1` (Optimal on TPU to prevent graph unrolling overhead)
- **HBM Footprint**: Micro-batch of 48 uses ~8.5 GB out of the 16 GB HBM per chip, avoiding OOM while keeping the 197 TFLOPS/chip compute units fully saturated.

---

### 4. XLA Graph Compilation Stability
- **Static Tensor Shapes**: Batch size, sequence length, and loss reduction shapes are strictly invariant across steps. This avoids XLA graph recompilations ("compilation storms").
- **Native bfloat16 AMP**: `torch.autocast(device_type="xla", dtype=torch.bfloat16)` utilizes native TPU BF16 hardware execution.
- **Fused SDPA**: `F.scaled_dot_product_attention` lowers directly to XLA's optimized TPU flash-attention kernel.

class WarmupCosineLR(torch.optim.lr_scheduler._LRScheduler):
    def __init__(self, optimizer, warmup_steps, max_steps, max_lr, min_lr=1e-5, last_epoch=-1):
        self.warmup_steps, self.max_steps, self.max_lr, self.min_lr = warmup_steps, max_steps, max_lr, min_lr
        super().__init__(optimizer, last_epoch)

    def get_lr(self):
        step = self.last_epoch
        if step < self.warmup_steps:
            lr = self.max_lr * (step + 1) / max(1, self.warmup_steps)
        elif step > self.max_steps:
            lr = self.min_lr
        else:
            decay_ratio = (step - self.warmup_steps) / max(1, self.max_steps - self.warmup_steps)
            lr = self.min_lr + 0.5 * (1.0 + math.cos(math.pi * decay_ratio)) * (self.max_lr - self.min_lr)
        return [float(lr) for _ in self.base_lrs]

# Paradigm Losses
def mdlm_loss_pytorch(model, clean_targets, mask_token_id=1, vocab_size=8192):
    B, T = clean_targets.shape
    t = torch.rand((B, 1), device=clean_targets.device)
    mask = torch.rand((B, T), device=clean_targets.device) < t
    corrupted = torch.where(mask, torch.full_like(clean_targets, mask_token_id), clean_targets)
    logits = model(corrupted)
    loss = nn.functional.cross_entropy(logits.view(-1, vocab_size), clean_targets.view(-1), reduction='none').view(B, T)
    safe_t = torch.clamp(t, min=1e-4)
    loss = (loss * mask.float() / safe_t).sum(dim=1)
    return loss.mean()

    t = torch.rand((B, 1), device=clean_targets.device)
    mask = torch.rand((B, T), device=clean_targets.device) < t
    random_tokens = torch.randint(0, vocab_size, (B, T), device=clean_targets.device)
    corrupted = torch.where(mask, random_tokens, clean_targets)

    layer_map = [min(int(i * (src_cfg["n_layers"] / tgt_cfg["n_layers"])), src_cfg["n_layers"] - 1) for i in range(tgt_cfg["n_layers"])]
    src_state = load_file(src_ckpt_path) if src_ckpt_path.endswith(".safetensors") else torch.load(src_ckpt_path, map_location="cpu")
    if "model_state_dict" in src_state: src_state = src_state["model_state_dict"]
    tgt_state = tgt_model.state_dict()
    for k, tgt_tensor in tgt_state.items():
        src_key = ".".join([k.split(".")[0], str(layer_map[int(k.split(".")[1])])] + k.split(".")[2:]) if "layers." in k else k
        if src_key in src_state:
            src_tensor = src_state[src_key]
            if tgt_tensor.shape == src_tensor.shape:
                tgt_tensor.copy_(src_tensor)
            elif "norm" in src_key and len(src_tensor.shape) == 1:

                slices = tuple(slice(0, min(s, t)) for s, t in zip(src_tensor.shape, tgt_tensor.shape))
                padded[slices] = src_tensor[slices]
                tgt_tensor.copy_(padded)

=== ORIGINAL STEP 837 | JSONL LINE 836 | MODEL | 2026-08-30T07:01:20Z ===
In PyTorch-XLA, operations are **lazy by default**. When you write `model(x)` and `loss.backward()`, PyTorch-XLA does **not** actually execute anything on the TPU hardware—it simply records mathematical nodes into an in-memory **Lazy Computation Graph (HLO)**.

`xm.mark_step()` is the **execution trigger and graph boundary**.

---

### What happens if you remove `xm.mark_step()`?

If you remove `xm.mark_step()` (and don't have explicit synchronization points like `loss.item()` or `xm.optimizer_step`):

#### 1. Host RAM Explosion (Unbounded Graph Accumulation)
Because the TPU hasn't been told where a training step ends, PyTorch-XLA will keep recording:
$$\text{Step 1} \rightarrow \text{Step 2} \rightarrow \text{Step 3} \dots \rightarrow \text{Step 500}$$
It chains hundreds of steps into **one massive, ever-expanding graph**. Your CPU host RAM will rapidly leak and trigger an OS OOM kill.

#### 2. XLA Compiler Freeze
When something *eventually* forces an evaluation (e.g., saving a checkpoint or evaluating validation loss), XLA will suddenly attempt to compile a single graph containing **thousands of unrolled steps**. 
The XLA compiler will hang for **15–45+ minutes** attempting graph optimization, dead-code elimination, and register scheduling on millions of concatenated operations.

#### 3. Immediate TPU HBM OOM
Because XLA views all accumulated steps as a single fused operation, it assumes all intermediate tensors across all steps must stay live simultaneously. The 16 GB TPU HBM will instantly crash on allocation.

---

### Summary: Why it’s in our script

1. **Defines Iteration Boundaries**: Tells the compiler *"This 1 step is the repeating template."* XLA compiles it **once** in the first step, caches the binary, and then runs at full hardware speed (~1M tok/s) for all remaining steps.
2. **Frees Intermediate Memory**: Allows the TPU to garbage-collect activation tensors at the end of each step.
3. **Pipeline Overlapping**: Allows TPU compute execution to overlap with host CPU preparation of the next micro-batch.

*(Note: `xm.optimizer_step(optimizer)` calls `xm.mark_step()` under the hood, but keeping an explicit `xm.mark_step()` at loop/micro-batch boundaries ensures the XLA graph is cut cleanly and never bleeds across iterations).*

=== ORIGINAL STEP 1161 | JSONL LINE 1142 | MODEL | 2026-08-30T18:21:45Z ===
The current PyTorch `corosred` implementation trains **two components jointly end-to-end**:

1. **The AR Language Model** (Backbone + `output_projection` LM Head):
   - Computes standard causal next-token prediction loss ($\mathcal{L}_{\text{AR}}$).
   - Trains the full transformer to draft high-quality token sequences.

2. **The Reliability Head** (`reliability_head`):
   - Computes Binary Cross-Entropy ($\mathcal{L}_{\text{BCE}}$) predicting whether each drafted token is clean ($1.0$) vs erroneous ($0.0$).
   - Uses **Ambiguity-Exclusion**: tokens that fall within the top-$k$ plausible alternatives ($k_{\text{amb}}=5$) are masked out from the loss so the reliability head doesn't get penalized on ambiguous valid syntax.

---

### What about the Refinement / Diffusion part (Phase B)?
In the full COROSred architecture:
- **Drafting & Flagging**: Uses the **Causal AR backbone + Reliability Head** (what is currently training).
- **Refinement**: Uses **Bidirectional Masked Denoising (MDLM)** to infill and fix the tokens flagged as low-reliability ($r_i < \tau$).

If you want the 100M model to also learn bidirectional masked denoising for the refinement pass, we can either:
1. **Joint Multi-Task Loss**: Add a bidirectional masked term ($\mathcal{L}_{\text{total}} = \mathcal{L}_{\text{AR}} + \mathcal{L}_{\text{Rel}} + \mathcal{L}_{\text{MDLM}}$) so the single model learns both causal drafting and bidirectional infilling.
2. **Use MDLM Checkpoint**: Keep them separate as originally designed (AR+Reliability Head drafts and flags, MDLM checkpoint refines).

Okay, so how do you plan on evaluating Phase A COROSred?

=== ORIGINAL STEP 1165 | JSONL LINE 1146 | MODEL | 2026-08-30T18:24:31Z ===
Phase A of COROSred evaluates whether the **Reliability Head** actually learns to detect errors and hallucinations better than standard softmax entropy. 

The evaluation framework (`COROSred/eval/evaluator.py`) runs **three core tests**:

---

### 1. Stratified ROC-AUC ($\Delta\text{AUC}$ vs Softmax Entropy)
Standard language models use output softmax entropy $H(p)$ as a crude uncertainty proxy. However, entropy fails on code where multiple valid branchings exist.
- **High-Entropy Stratum ($H(p) \ge 1.5$)**: 
  - Measures ROC-AUC of the Reliability Head predicting correctness on high-entropy tokens vs raw negative entropy.
  - **Gate Criteria**: $\Delta\text{AUC} = \text{AUC}_{\text{Head}} - \text{AUC}_{\text{Entropy}} \ge +0.05$. (Proves the head learned genuine error calibration beyond trivial entropy).

---

### 2. "Confidently Wrong" Recall (Low-Entropy Stratum)
- **Low-Entropy Stratum ($H(p) < 1.5$)**:
  - Tests cases where the model hallucinates with high confidence (low entropy, but wrong token).
  - **Metric**: Percentage of confidently wrong predictions flagged with negative reliability scores ($r_i < 0$).

---

### 3. Synthetic Corruption Localization (Out-of-Distribution Error Detection)
- Injects random synthetic corruption tokens into clean test code ($p = 0.05$).
- Evaluates whether the reliability head flags the exact corrupted token positions without having seen the corruptions during training.
- **Metric**: Full sequence token-level Localization ROC-AUC.

---

### 4. Downstream Code Quality (Standard AR Baseline)
- Standard validation perplexity ($\text{PPL}$) and syntax accuracy on the test corpus to ensure training the joint reliability objective did not degrade AR generation capabilities.

### Root Cause
1. **Gradient Conflict at `final_norm`**:
   In our joint loss ($\mathcal{L}_{\text{total}} = \mathcal{L}_{\text{AR}} + \mathcal{L}_{\text{BCE}}$), the `reliability_head` backpropagated directly into the shared hidden states $h$ from Step 0. 
   At cold initialization (accuracy $\approx 0.01\%$), the BCE loss pushed massive gradients through `final_norm` telling the backbone that all tokens are unconfident, directly fighting the Autoregressive Cross-Entropy loss trying to make tokens confident. This caused gradient explosion at `final_norm`.
2. **Phase A Architecture Requirement**:
   In the original COROSred specification, Phase A trains the Reliability Head on the representations without letting the BCE loss corrupt the AR backbone. Passing `h.detach()` to `reliability_head` completely isolates the backbone for clean, stable AR training while the reliability head learns token calibration.
